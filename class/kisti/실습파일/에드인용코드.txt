Sub 회색헤더()

    Dim selectedRange As Range
    
    ' 선택한 범위 확인
    On Error Resume Next
    Set selectedRange = Selection
    On Error GoTo 0
    
    ' 선택한 범위가 없는 경우 메시지 출력 후 종료
    If selectedRange Is Nothing Then
        MsgBox "범위를 선택하세요.", vbExclamation
        Exit Sub
    End If

    ' 전체 내부 테두리
    With selectedRange.Borders
        .LineStyle = xlContinuous
        .Color = vbBlack
        .Weight = xlThin
    End With
    
    ' 바깥쪽 테두리 굵게
    With selectedRange
        .Borders(xlEdgeLeft).Weight = xlThick
        .Borders(xlEdgeTop).Weight = xlThick
        .Borders(xlEdgeRight).Weight = xlThick
        .Borders(xlEdgeBottom).Weight = xlThick
    End With

    ' 헤더 스타일
    selectedRange.Rows(1).Interior.Color = RGB(200, 200, 200)
    selectedRange.Rows(1).Borders(xlEdgeBottom).LineStyle = xlDouble

    MsgBox "완료되었습니다.", vbInformation

End Sub
Sub 동일셀병합()

    Dim ws As Worksheet
    Dim rng As Range
    Dim arr As Variant
    
    Dim r As Long
    Dim c As Long
    Dim startR As Long
    Dim endR As Long
    
    Dim rowCount As Long
    Dim colCount As Long
    
    Dim calcMode As XlCalculation
    
    Set ws = ActiveSheet
    
    If TypeName(Selection) <> "Range" Then
        MsgBox "셀 범위를 선택한 후 실행하세요.", vbExclamation
        Exit Sub
    End If
    
    Set rng = Selection
    
    ' 다중 영역 선택 방지
    If rng.Areas.Count > 1 Then
        MsgBox "연속된 하나의 범위만 선택하세요.", vbExclamation
        Exit Sub
    End If
    
    ' 최소 2행 이상 필요 (첫 행은 헤더 제외)
    If rng.Rows.Count < 2 Then
        MsgBox "헤더 포함 2행 이상 선택하세요.", vbExclamation
        Exit Sub
    End If
    
    calcMode = Application.Calculation
    
    On Error GoTo EH
    
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    Application.Calculation = xlCalculationManual
    Application.EnableEvents = False
    
    arr = rng.Value2
    
    rowCount = rng.Rows.Count
    colCount = rng.Columns.Count
    
    ' 첫 행은 헤더로 가정 → 2행부터 시작
    For c = 1 To colCount
        
        r = 2
        
        Do While r <= rowCount
            
            startR = r
            
            Do While r < rowCount
                ' 오류값이 있으면 비교 중단
                If IsError(arr(r, c)) Or IsError(arr(r + 1, c)) Then Exit Do
                
                ' 빈 셀은 병합하지 않도록 처리
                If Trim(CStr(arr(startR, c))) = "" Then Exit Do
                
                ' 다음 행과 값이 같으면 계속 진행
                If CStr(arr(r + 1, c)) = CStr(arr(startR, c)) Then
                    r = r + 1
                Else
                    Exit Do
                End If
            Loop
            
            endR = r
            
            If endR > startR Then
                With ws.Range(rng.Cells(startR, c), rng.Cells(endR, c))
                    .merge
                    .HorizontalAlignment = xlCenter
                    .VerticalAlignment = xlCenter
                End With
            End If
            
            r = r + 1
            
        Loop
        
    Next c
    
CleanExit:
    Application.EnableEvents = True
    Application.Calculation = calcMode
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True
    MsgBox "선택된 영역의 상하행 데이터가 동일한 셀의 병합이 완료되었습니다.", vbInformation
    Exit Sub

EH:
    MsgBox "오류가 발생했습니다: " & Err.Number & vbCrLf & Err.Description, vbExclamation
    Resume CleanExit

End Sub


Sub 병합해제()
    Dim ws As Worksheet
    Dim rng As Range
    Dim cell As Range
    Dim lastRow As Long
    Dim lastCol As Long
    Dim r As Long, c As Long
    
    ' 현재 작업 중인 워크시트를 설정
    Set ws = ActiveSheet
    
    ' 사용된 범위를 기준으로 마지막 행과 열 계산
    With ws.UsedRange
        lastRow = .Rows(.Rows.Count).Row
        lastCol = .Columns(.Columns.Count).Column
    End With
    
    ' 사용된 범위에서 병합된 셀 해제
    On Error Resume Next ' 오류가 생겨도 무시하고 진행
    ws.UsedRange.UnMerge
    On Error GoTo 0 ' 오류 처리 초기화
    
    ' 각 셀을 순회하면서 빈 셀을 윗 셀의 값으로 채우기
    For c = 1 To lastCol
        For r = 2 To lastRow ' 1행은 제목일 가능성이 있어 2행부터 시작
            Set cell = ws.Cells(r, c)
            If IsEmpty(cell.Value) Then
                cell.Value = ws.Cells(r - 1, c).Value
            End If
        Next r
    Next c
    
    ' 전체 사용 범위에 테두리 적용
    Set rng = ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, lastCol))
    
    With rng.Borders
        .LineStyle = xlContinuous
        .Color = vbBlack
        .Weight = xlThin
    End With
    
    MsgBox "병합 해제가 완료되었습니다.", vbInformation
End Sub
Public Sub 공백제거()

    Dim c As Range
    Dim txt As String
    
    If TypeName(Selection) <> "Range" Then Exit Sub
    
    For Each c In Selection
    
        If Not IsError(c.Value) Then
        
            txt = CStr(c.Value)
            
            '줄바꿈 제거
            txt = Replace(txt, vbCr, " ")
            txt = Replace(txt, vbLf, " ")
            
            'Trim
            txt = Application.WorksheetFunction.Trim(txt)
            
            '연속 공백 제거
            Do While InStr(txt, "  ") > 0
                txt = Replace(txt, "  ", " ")
            Loop
            
            c.Value = txt
            
        End If
        
    Next c
    
    MsgBox "정리가 완료되었습니다."

End Sub
Sub 파일합치기()
    Dim fd As FileDialog
    Dim mainWb As Workbook, sourceWb As Workbook
    Dim mainWs As Worksheet, sourceWs As Worksheet
    Dim lastRow As Long, nextRow As Long
    Dim isFirstFile As Boolean
    Dim i As Integer
    Dim firstFilePath As String, firstFileName As String
    Dim saveFolder As String, savePath As String
    
    ' 1. 화면 업데이트 및 경고 메시지 비활성화 (속도 향상)
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False
    
    ' 2. 다중 파일 선택 창 실행 (msoFileDialogFilePicker = 파일 선택, 복수 선택 허용)
    Set fd = Application.FileDialog(msoFileDialogFilePicker)
    fd.AllowMultiSelect = True                        ' 다중 파일 선택 활성화
    fd.Filters.Clear                                  ' 기존 필터 초기화
    fd.Filters.Add "Excel 파일", "*.xlsx; *.xlsm; *.xls"  ' xlsx/xlsm/xls만 표시
    fd.Title = "병합할 Excel 파일을 선택하세요 (다중 선택 가능)"
    
    ' 2-1. 파일 선택 취소 시 종료
    If fd.Show <> -1 Then
        MsgBox "파일이 선택되지 않았습니다. 종료합니다."
        Application.ScreenUpdating = True
        Application.DisplayAlerts = True
        Exit Sub
    End If
    
    ' 2-2. 선택된 파일이 없을 경우 예외 처리
    If fd.SelectedItems.Count = 0 Then
        MsgBox "선택된 파일이 없습니다. 종료합니다."
        Application.ScreenUpdating = True
        Application.DisplayAlerts = True
        Exit Sub
    End If
    
    ' 3. 첫 번째 파일 경로에서 [폴더 경로]와 [파일명(확장자 제외)] 분리
    '    - firstFilePath : "C:\Users\data\매출_1월.xlsx"
    '    - saveFolder    : "C:\Users\data\"
    '    - firstFileName : "매출_1월"  (확장자 제거)
    firstFilePath = fd.SelectedItems(1)
    saveFolder = Left(firstFilePath, InStrRev(firstFilePath, "\"))  ' 마지막 "\" 이전까지 = 폴더 경로
    firstFileName = Mid(firstFilePath, InStrRev(firstFilePath, "\") + 1)  ' 파일명(확장자 포함)
    firstFileName = Left(firstFileName, InStrRev(firstFileName, ".") - 1) ' 확장자 제거
    
    ' 4. 최종 저장 경로 조합: {폴더경로}\{첫번째파일명}_파일병합.xlsx
    savePath = saveFolder & firstFileName & "_파일병합.xlsx"
    
    ' 5. 결과 데이터를 담을 새 통합 문서 및 시트 생성
    Set mainWb = Workbooks.Add
    Set mainWs = mainWb.Sheets(1)
    isFirstFile = True
    
    ' 6. 선택된 파일 목록을 순서대로 순회 (1번 ~ 마지막 파일)
    For i = 1 To fd.SelectedItems.Count
        
        ' 6-1. 현재 순번의 파일 열기 (읽기 전용)
        Set sourceWb = Workbooks.Open(fd.SelectedItems(i), ReadOnly:=True)
        Set sourceWs = sourceWb.Sheets(1)
        
        ' 6-2. 소스 파일의 마지막 데이터 행 계산 (A열 기준)
        lastRow = sourceWs.Cells(sourceWs.Rows.Count, "A").End(xlUp).Row
        
        ' 6-3. 데이터가 존재하는 경우에만 복사 작업 수행 (1행만 있는 경우 = 헤더만 있는 경우 제외)
        If lastRow > 1 Then
            If isFirstFile Then
                ' 7. 첫 번째 파일: 헤더 포함 전체 복사 → mainWs A1 셀부터 붙여넣기
                sourceWs.UsedRange.Copy mainWs.Range("A1")
                isFirstFile = False
            Else
                ' 8. 두 번째 파일부터: 헤더(1행) 제외하고 2행~마지막행만 복사
                '    → mainWs의 현재 마지막 행 다음 행부터 이어붙이기
                nextRow = mainWs.Cells(mainWs.Rows.Count, "A").End(xlUp).Row + 1
                sourceWs.Range("A2", sourceWs.Cells(lastRow, sourceWs.UsedRange.Columns.Count)).Copy _
                    mainWs.Range("A" & nextRow)
            End If
        End If
        
        ' 9. 소스 파일 닫기 (저장 없이)
        sourceWb.Close SaveChanges:=False
        
    Next i  ' 다음 파일로 이동
    
    ' 10. 데이터 정렬 (A열 기준 오름차순)
    With mainWs.Sort
        .SortFields.Clear
        .SortFields.Add2 Key:=mainWs.Range("A1"), Order:=xlAscending
        .SetRange mainWs.UsedRange
        .Header = xlYes
        .Apply
    End With
    
    ' 11. 서식 설정: 열 폭 자동 맞춤
    mainWs.UsedRange.Columns.AutoFit
    
    ' 12. 서식 설정: 헤더 배경색 (하늘색, RGB: 135, 206, 235)
    Dim headerRange As Range
    Set headerRange = mainWs.Range(mainWs.Cells(1, 1), mainWs.Cells(1, mainWs.UsedRange.Columns.Count))
    headerRange.Interior.Color = RGB(135, 206, 235)
    
    ' 13. 동일 파일명 존재 시 자동으로 _1, _2... 번호 붙이기
    '     - basePath  : 확장자 제거한 경로 ("C:\data\매출_1월_파일병합")
    '     - finalPath : 최종 저장 경로 (번호 포함 가능)
    Dim basePath As String
    Dim finalPath As String
    Dim counter As Integer
    
    basePath = Left(savePath, InStrRev(savePath, ".") - 1)  ' 확장자 제거 ("..._파일병합")
    finalPath = savePath                                     ' 우선 원본 경로로 시작
    counter = 1
    
    ' 13-1. 파일이 존재하는 동안 계속 번호를 증가시키며 새 경로 생성
    '        Dir() 함수가 빈 문자열을 반환하면 = 해당 경로에 파일 없음 = 사용 가능
    Do While Dir(finalPath) <> ""
        finalPath = basePath & "_" & counter & ".xlsx"  ' 예: 매출_파일병합_1.xlsx
        counter = counter + 1
    Loop
    
    ' 13-2. 최종 확정된 경로에 파일 저장
    mainWb.SaveAs Filename:=finalPath, FileFormat:=51
    mainWb.Close
    
    ' 14. 설정 복구
    Application.ScreenUpdating = True
    Application.DisplayAlerts = True
    
    MsgBox "작업이 완료되었습니다." & vbCrLf & "저장 경로: " & savePath
End Sub

Sub 시트나누기()

    Dim wsSource As Worksheet, wsNew As Worksheet
    Dim rngData As Range
    Dim colValues As New Collection
    Dim cell As Range
    
    Dim lastRow As Long
    Dim splitCol As Long
    Dim inputCol As String
    
    Dim i As Long

    ' 기준 열 입력
    inputCol = InputBox("기준 열을 입력하세요 (예: B 또는 2)", "시트 분리 기준 열")
    
    If inputCol = "" Then Exit Sub
    
    If IsNumeric(inputCol) Then
        splitCol = CLng(inputCol)
    Else
        splitCol = Range(UCase(inputCol) & "1").Column
    End If

    Set wsSource = ActiveSheet
    
    lastRow = wsSource.Cells(wsSource.Rows.Count, splitCol).End(xlUp).Row
    Set rngData = wsSource.Range("A1").CurrentRegion

    Application.ScreenUpdating = False
    Application.DisplayAlerts = False

    ' 고유값 추출
    On Error Resume Next
    For Each cell In wsSource.Range(wsSource.Cells(2, splitCol), wsSource.Cells(lastRow, splitCol))
        If cell.Value <> "" Then
            colValues.Add cell.Value, CStr(cell.Value)
        End If
    Next cell
    On Error GoTo 0

    ' 시트 분리
    For i = 1 To colValues.Count
        
        Set wsNew = Worksheets.Add(After:=Worksheets(Worksheets.Count))
        wsNew.Name = Left(colValues(i), 31)
        
        wsSource.AutoFilterMode = False
        rngData.AutoFilter Field:=splitCol, Criteria1:=colValues(i)
        
        rngData.SpecialCells(xlCellTypeVisible).Copy
        
        wsNew.Range("A1").PasteSpecial xlPasteAll
        wsNew.Range("A1").PasteSpecial xlPasteColumnWidths
        
    Next i

    wsSource.AutoFilterMode = False

    Application.ScreenUpdating = True
    Application.DisplayAlerts = True

    MsgBox "시트 분리가 완료되었습니다."

End Sub
Sub 파일나누기()
    Dim ws As Worksheet
    Dim fso As Object
    Dim mainPath As String, localPath As String
    Dim folderPath As String, newFileName As String
    
    Set fso = CreateObject("Scripting.FileSystemObject")
    mainPath = ThisWorkbook.Path
    
    ' 1. 파일 저장 여부 확인
    If mainPath = "" Then
        MsgBox "파일을 먼저 저장한 후 실행해 주세요.", vbCritical
        Exit Sub
    End If

    ' 2. OneDrive/SharePoint 웹 경로인지 확인 및 로컬 경로 변환
    ' 웹 경로(https://)인 경우 환경 변수를 이용해 로컬 경로를 추정합니다.
    If Left(mainPath, 4) = "http" Then
        ' 사용자 환경의 OneDrive 로컬 경로를 가져옵니다.
        localPath = Environ("OneDrive")
        If localPath = "" Then localPath = Environ("OneDriveCommercial")
        
        ' 실제 로컬 폴더 경로 구성 (사용자 환경에 따라 조정이 필요할 수 있음)
        ' 파일이 저장된 폴더 이름만 추출하여 로컬 경로와 합칩니다.
        folderPath = localPath & "\거래처별자료"
    Else
        ' 일반 로컬 경로인 경우
        folderPath = mainPath & "\거래처별자료"
    End If

    ' 3. 폴더 존재 확인 및 생성
    If Not fso.FolderExists(folderPath) Then
        On Error Resume Next
        fso.CreateFolder folderPath
        If Err.Number <> 0 Then
            MsgBox "폴더를 생성할 수 없습니다. 경로를 확인해 주세요:" & vbCrLf & folderPath, vbCritical
            Exit Sub
        End If
        On Error GoTo 0
    End If
    
    Application.ScreenUpdating = False
    Application.DisplayAlerts = False

    ' 4. 시트 순환 및 저장
    For Each ws In ThisWorkbook.Worksheets
        If ws.Visible = xlSheetVisible Then
            ws.Copy
            newFileName = folderPath & "\" & ws.Name & ".xlsx"
            
            ' 파일 저장 (파일 이름에 특수문자가 있으면 오류가 날 수 있음)
            ActiveWorkbook.SaveAs Filename:=newFileName, FileFormat:=51
            ActiveWorkbook.Close SaveChanges:=False
        End If
    Next ws

    Application.DisplayAlerts = True
    Application.ScreenUpdating = True

    MsgBox "작업 완료: " & folderPath & " 폴더를 확인하세요.", vbInformation
    Set fso = Nothing
End Sub
